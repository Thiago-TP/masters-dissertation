\begin{frame}{Methods}{Metrics}
    Four metrics are considered on the evaluation of posterior/prior distributions
    \cite[subsection 9.4.2]{Emerick:2025}:
        history observation coverage,
        validation observation coverage,
        mean squared error, 
        and data-mismatch.
        Denoting an ensemble by $\mathbf{Y} = \begin{bmatrix}
            \mathbf{y}_1 & \cdots & \mathbf{y}_{N_e}
        \end{bmatrix}$, these metrics are defined as:
    %
    \begin{align}
        \text{OC}_h(\mathbf{Y}) &= \frac{1}{N_h} \sum_{i=1}^{N_h} 
        \mathbbm{1}\left(
            \min(\mathbf{Y})_i \leq d_{h, i} \leq \max(\mathbf{Y})_i
        \right)\\
        \text{OC}_v(\mathbf{Y}) &= \frac{1}{N_v} \sum_{i=1}^{N_v} 
        \mathbbm{1}\left(
            \min(\mathbf{Y})_i \leq d_{v, i} \leq \max(\mathbf{Y})_i
        \right)\\
        %
        \text{MSE}(\mathbf{Y}) &= 
        \frac{1}{N_h} 
        (\mathbf{d}_h - \overline{\mathbf{y}})^\top \mathbf{C}^{-1} 
        (\mathbf{d}_h - \overline{\mathbf{y}}),
        \quad 
        \mathbf{C} = \Delta\mathbf{Y}\Delta\mathbf{Y}^\top + \mathbf{C}_e \\
        %
        \mathcal{O}(\mathbf{y}_j) &= 
        \frac{1}{2N_h}
        (\mathbf{d}_h - \mathbf{y}_{j, h})^\top
        \mathbf{C}_e^{-1}
        (\mathbf{d}_h - \mathbf{y}_{j, h})
    \end{align}
\end{frame}
