\section{Theory framework}

\begin{frame}{Theory framework}{Important concepts and notation}
    \begin{block}{Prediction forward model}
        The prediction forward model, or forward model, 
        is the underlying mapping between a  model's parameters, controls and initial states to its observable features.
    \end{block}
    %
    \vfill
    %
    Denoting by $\mathbf{m}$ the vector of model parameters,
    and by $\mathbf{u}$ the set of controls applied on the field,
    and by $\mathbf{d}$ the reservoir production within the time frame of the controls,
    one can write
    %
    \begin{equation}
        \mathbf{d} = \mathbf{g}(\mathbf{m}, \mathbf{u}; \mathbf{x}_0)
    \label{eq:prediction-forward-model}
    \end{equation} 
    %
    with $\mathbf{x}_0$ the initial value of reservoir states (usually omitted).
    %
    \vfill
\end{frame}

\begin{frame}{Theory framework}{Important concepts and notation}
    \begin{exampleblock}{Data features}
        In this work, the features inside $\mathbf{d}$ are the following well measurements:

        \medskip 

        \def\arraystretch{1.5}
        \begin{tabular}{r l l l}
            $\triangleright$ & oil and water production rates & (OPR and WPR) & [m$^3$/day] \\
            $\triangleright$ & water injection rate           & (WIR)         & [m$^3$/day] \\
            $\triangleright$ & bottomhole pressure            & (BHP)         & [Bar]       \\
            $\triangleright$ & water cut percentage           & (WCTP)        & [\%]

    \vfill

    \begin{alertblock}{Field oil production total}
        Field oil production total (FOPT, SI unit of m$^3$)
        is also relevant to this work, despite not being part of $\mathbf{d}$.
    \end{alertblock}
\end{frame}

\begin{frame}{Theory framework}{Important concepts and notation}
    The key assumption behind the forward model is that it is deterministic. 
    Implementation wise, $\mathbf{g}$ consists of evaluating a set of equations 
    at each gridblock of the model.
    %
    \vfill
    %
    Each evaluation evolves the model a little further in time, 
    until the scheduled time period is appreciated, concluding the process.
    %
    \vfill
    %
    Black-oil equations are the most common implementations of the forward model
    \cite{Rasmussen:2021} due to their fair simplicity and accuracy.
    The fluid is assumed composed of three liquid phases, water, oil and gas.
    %
    \vfill
\end{frame}

\begin{frame}{Theory framework}{Important concepts and notation}
    A given phase $\alpha$'s component mass $A_\alpha$ satisfies the conservation equation
    %
    \begin{align}
        \frac{\partial}{\partial t} \left(\phi_{\text{ref}} A_\alpha \right)
        + \nabla \cdot \mathbf{u}_\alpha + q_\alpha & = 0,
    \label{eq:materialbalance-cont}
    \end{align}
    %
    where 
    %
    \begin{subequations}
        \label{eq:acc}
        \begin{align}
            A_w &= m_{\phi} b_{w} s_{w},                      & \mathbf{u}_w &= b_{w} \mathbf{v}_{w}, \\
            A_o &= m_{\phi} (b_{o} s_{o} + r_{og}b_{g}s_{g}), & \mathbf{u}_o &= b_{o} \mathbf{v}_{o} +
            r_{og}b_{g}\mathbf{v}_{g}, \\
            A_g &= m_{\phi} (b_{g} s_{g} + r_{go}b_{o}s_{o}), & \mathbf{u}_g &= b_{g} \mathbf{v}_{g} +
            r_{go}b_{o}\mathbf{v}_{o}.
        \end{align}
    \end{subequations}
    %
    and the phase fluxes are given by Darcy's law:
    %
    \begin{align}
        \mathbf{v}_\alpha = -\lambda_\alpha \mathbf{K} (\nabla p_\alpha - \rho_\alpha \mathbf{g}).
        \label{eq:Darcy-mph}
    \end{align}
    %
    Finally, the following closure equations are imposed:
    %
    \begin{align}
        s_w + s_o + s_g &= 1,
        &p_{c,ow} &= p_{o} - p_{w},
        &p_{c,og} &= p_{o} - p_{g}. \label{eq:capillary}
    \end{align} 
\end{frame}

\begin{frame}{Theory framework}{Important concepts and notation}
    Let $\mathbf{d}$ denote the true production data vector of a reference reservoir across time,
    free of observation error.
    %
    Data is separated into a history period and a validation period,
    so that 
    \begin{equation}
        \mathbf{d} = 
        \begin{bmatrix}
            \mathbf{d}_h \\
            \mathbf{d}_v
        \end{bmatrix}
    \end{equation}
    %
    with $\mathbf{d}_h$ the ground-truth history data, and $\mathbf{d}_v$ the ground-truth validation data.
    %
    \vfill
    %
    Ground-truth data come from the forward model,
    %
    \begin{equation}
        \mathbf{d} = \mathbf{g}(\mathbf{m}, \mathbf{u}).
    \end{equation}
    %
    and observed data, from a corruption of $\mathbf{d}_h$:
    %
    \begin{equation}
        \mathbf{d}_\text{obs} = \mathbf{d}_h + \mathcal{N}(\mathbf{0}, \mathbf{C}_e),
    \end{equation}
    %
    where $\mathbf{C}_e$ is the noise covariance matrix.
\end{frame}

\begin{frame}{Theory framework}{Statistical framework}
    DSI-ESMDA can be conceptulized as a Bayesian approach to the forecasting problem,
    aiming to estimate the underlying conditional probability distribution of 
    predictions given the reference data.
    %
    \vfill
    %
    \begin{alertblock}{The DSI-ESMDA problem}
        Find the posterior distribution
        %
        \begin{equation}
            p(\mathbf{d}^*\mid \mathbf{d}_\text{obs}) \propto
            \mathcal{L}(\mathbf{d}_\text{obs} \mid \mathbf{d}^*) p(\mathbf{d}^*).
        \end{equation}
        %
        where $\mathbf{d}^*$ maximizes $p(\mathbf{d}^*\mid \mathbf{d}_\text{obs})$.
    \end{alertblock}
    %
    \vfill
    %
    The algorithm returns samples from this posterior distribution,
    outputing a range of values rather than a single value.
    %
    While this makes evaluating the algorithm less straightforward, 
    it allows for quantification of uncertainty.
\end{frame}

\begin{frame}{Theory framework}{Statistical framework}
    Ensemble Smoothers share this approach at a conceptual level, 
    but they perform a single maximization.
    %
    This leads to poor matches to real data if assumptions on the likelihood $\mathcal{L}$ are not met.

    \medskip

    Multiple Data Assimilation (MDA) mitigates this problem by making $N_a$ optimizations with resampled reference data: 
    %
    \begin{align}
        {\color{darkred}p^1(\mathbf{d}^*\mid \mathbf{d}^1_\text{obs})} 
        &\propto \mathcal{L}(\mathbf{d}^1_\text{obs}\mid \mathbf{d}^*) 
                            p(\mathbf{d}^*) 
    \label{eq:p1}
        \\
        p^2(\mathbf{d}^*\mid \mathbf{d}^2_\text{obs}, \mathbf{d}^1_\text{obs}) 
        &\propto \mathcal{L}(\mathbf{d}^2_\text{obs}\mid \mathbf{d}^*) 
                            {\color{darkred}p^1(\mathbf{d}^*\mid \mathbf{d}^1_\text{obs})} 
    \label{eq:p2} 
        \\
        &\ \ \vdots 
        \\
        p^{N_a}(\mathbf{d}^*\mid \mathbf{d}^{N_a}_\text{obs}, \dots, \mathbf{d}^1_\text{obs})
        &\propto \mathcal{L}(\mathbf{d}^{N_a}_\text{obs} \mid \mathbf{d}^*)
            {\color{darkred}p^{N_a-1}(\mathbf{d}^*\mid \mathbf{d}^{N_a-1}_\text{obs}, \dots, \mathbf{d}^1_\text{obs})} 
    \label{eq:pNa} 
    \end{align}  
\end{frame}

\begin{frame}{Theory framework}{The base equation}
    MDA makes the following suppositions on the likelihoods:
    \begin{itemize}
        \item $\mathcal{L}^\ell$ is Gaussian
        \item $\mathcal{L}^\ell$ has covariance 
              $\mathbf{C}^\ell_e = \alpha^\ell \mathbf{C}_e$, $\alpha^\ell > 0$
    \end{itemize}
    %
    This leads to 
    %
    \begin{equation}
        \mathcal{L}^\ell(\mathbf{d}^\ell_\text{obs} \mid \mathbf{d}^*)
        \propto 
        \exp\left(-
            \frac{1}{2} 
            (\mathbf{d}_\text{obs} - \mathbf{H}\mathbf{d}^*)^\top 
            \frac{\mathbf{C}_e^{-1}}{\alpha^\ell}
            (\mathbf{d}_\text{obs} - \mathbf{H}\mathbf{d}^*)
        \right),
    \end{equation}
    %
    $\mathbf{H}$ a matrix that extracts $\mathbf{d}^*_h$ from $\mathbf{d}^*$.
    %
    \vfill
    %
    \begin{block}{Equivalence to ES}
        For Gaussian likelihoods, MDA is equivalent to ES if 
        \begin{equation}
            \sum_{\ell=1}^{N_a} \frac{1}{\alpha^\ell} = 1,
        \end{equation}
        highlighting the importance of having inflation factors on the covariance.
    \end{block}
\end{frame}

\begin{frame}{Theory framework}{The base equation}
    The priors are also supposed Gaussian:
    %
    \begin{equation}
        p^\ell(\mathbf{d}^* \mid \mathbf{d}^\ell_\text{prior}) \propto \exp\left(-
            \frac{1}{2}
            (\mathbf{d}^* - \mathbf{d}_\text{prior})^\top 
            (\mathbf{C}^\ell_p)^{-1}
            (\mathbf{d}^* - \mathbf{d}_\text{prior})
        \right),
    \end{equation}
    %
    where 
    
    \medskip

    \def\arraystretch{1.5}
    \begin{tabular}{lll}
        $\triangleright$ & $\mathbf{d}^\ell_\text{prior}$ 
            & is the prediction with highest belief of occurring prior to assimilating $\mathbf{d}^\ell_\text{obs}$\\
        $\triangleright$ & $\mathbf{C}^\ell_p$ & is the prior covariance matrix (i.e., $\mathbf{d}^* \sim \mathcal{N}(\mathbf{d}^\ell_\text{prior}, \mathbf{C}^\ell_p)$)\\ 
    \end{tabular}
\end{frame}

\begin{frame}{Theory framework}{The base equation}
    The maximum \emph{a posteriori} (MAP) estimate of the $\ell$-th assimilation becomes
    %
    \begin{subequations}
        \begin{align}
            \mathbf{d}^{\ell+1}_\text{map}
            &=
            \mathbf{d}^\ell_\text{map} +
            \mathbf{K}^\ell (\mathbf{d}^\ell_\text{obs} - \mathbf{d}^\ell_{h, \text{map}}), 
        \\
            \mathbf{K}^\ell 
            &= 
            \mathbf{C}^\ell_p \mathbf{H}^\top 
            (\mathbf{H}\mathbf{C}_p^\ell \mathbf{H}^\top + \alpha^\ell\mathbf{C}_e)^{-1}
        \end{align}
    \end{subequations}
    %
    with $\mathbf{d}^0_\text{map} = \mathbf{d}^0_\text{prior}$.
    %
    \vfill
    %
    Using an ensemble of $N_e$ model realizations, one can estimate the prior covariances with 
    $\Delta\mathbf{D} = \begin{bmatrix}
        \mathbf{d}_1 - \overline{\mathbf{d}} & \cdots & 
        \mathbf{d}_{N_e} - \overline{\mathbf{d}}
    \end{bmatrix}$:
    %
    \begin{subequations}
        \begin{alignat}{4}
            \mathbf{d}_j^{\ell+1} 
            &=
            \mathbf{d}_j^\ell + \mathbf{K}^\ell
            (\mathbf{d}_\text{obs} + \sqrt{\alpha^\ell}\mathbf{e}_j^\ell - \mathbf{d}_{h,j}^\ell),
            \\
            \mathbf{K}^\ell 
            &=  
            \Delta\mathbf{D}^\ell\Delta\mathbf{D}^\ell_h
            (\Delta\mathbf{D}^\ell_h\Delta\mathbf{D}^\ell_h + \alpha^\ell\mathbf{C}_e)^{-1}.
            \label{eq:kalman-gain-l}
        \end{alignat}
    \label{eq:iteration_dsiesmda-no-loc}
    \end{subequations}

    $\mathbf{d}_\text{obs} + \sqrt{\alpha^\ell}\mathbf{e}_j^\ell$, 
    with $\mathbf{e}_j\sim\mathcal{N}(\mathbf{0}, \mathbf{C}_e)$,
    is a way of sampling 
    $\mathbf{d}_\text{obs}^\ell \sim 
    \mathcal{N}(\mathbf{d}_\text{obs}, \alpha^\ell\mathbf{C}_e)$. 
\end{frame}

\begin{frame}{Theory framework}{Covariance localization}
    Covariance localization, or simply localization, 
    is an \emph{ad hoc} procedure employed to mitigate the effects of small ensembles 
    \cite[section 7.5]{Emerick:2025}.
    %
    \vfill
    %
    The influence a datum has on another datum is weighted
    in a inverse manner to the distance between the points.
    %
    \vfill
    %
    Localization is applied to the Kalman-like gain,
    so that the base equation becomes:
    %
    \begin{equation}
        \mathbf{d}_j^{\ell+1} 
        =
        \mathbf{d}_j^\ell + (\mathbf{R}\circ\mathbf{K}^\ell)
        (\mathbf{d}_\text{obs} + \sqrt{\alpha^\ell}\mathbf{e}_j^\ell - \mathbf{d}_{h, j}^\ell),
    \label{eq:iteration_dsiesmda}
    \end{equation}
\end{frame}

\begin{frame}{Theory framework}{Proposal}
    This work introduces gating $\bm{\alpha}$ on the innovation vector:  

    \begin{subequations}
        \begin{align}
            \mathbf{d}_j^{\ell+1} 
            &=
            \mathbf{d}_j^\ell + (\mathbf{R} \circ \mathbf{K}^\ell) 
            [
            \bm{\alpha}_j^\ell \circ 
            (\mathbf{d}_\text{obs} - \mathbf{d}_{h, j}^\ell)], 
        \label{eq:iteration_gated}
            \\
            \alpha_{j, v}^\ell 
            &= \mathrm{erf}
            \left(
                \frac{1}{\lambda \sqrt{2}}
                \frac{|(\mathbf{d}_\text{obs} - \mathbf{d}_{h, j}^\ell)_v|}{\sigma_v} 
            \right).
        \label{eq:alpha_lambda}
        \end{align}
    \end{subequations}
    
    The gating recovers variance loss in the posterior ensemble by diminishing Kalman-like updates on reasonably matched data.
    The smaller the mismatch of a measurement relative to the noise at that datum, the smaller the update.
